import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

```bash
composer require nwidart/laravel-metrics
```

<CardGrid>
	<Card title="Zero-code instrumentation" icon="rocket">
		`http.server.request.duration`, `db.client.operation.duration`, `messaging.process.duration`,
		cache operations, scheduled tasks and reported exceptions, with the names of the
		OpenTelemetry semantic conventions.
	</Card>
	<Card title="Built for PHP-FPM" icon="setting">
		Each request records in memory, then merges into Redis after the response is sent. Every
		worker and every host adds up in one store.
	</Card>
	<Card title="Prometheus or OTLP" icon="external">
		Serve `GET /metrics` in the Prometheus text format, or push to an OpenTelemetry Collector or
		any OTLP backend with `php artisan metrics:push`.
	</Card>
	<Card title="Safe by default" icon="approve-check">
		A Redis failure never fails a request. Cardinality is bounded, and the scrape route is off
		until you turn it on.
	</Card>
</CardGrid>
